```python
"""
Модуль для загрузки параметров программы
=========================================================================================

Этот модуль загружает конфигурационные параметры из файла 'config.json'
с использованием функции `j_loads_ns`.

Пример использования
--------------------

.. code-block:: python

    from src.gs import gs

    print(gs)
"""
from __future__ import annotations

# -*- coding: utf-8 -*-
#! .pyenv/bin/python3
from pathlib import Path
from typing import Dict, Any

from src.utils.jjson import j_loads_ns
from src.logger.logger import logger


try:
    # Код загружает конфигурацию из файла 'config.json' с использованием 'j_loads_ns'
    gs: Dict[str, Any] = j_loads_ns(Path('config.json'))
except Exception as e:
    # Логирование ошибки, если не удалось загрузить конфигурационный файл
    logger.error(f'Ошибка при загрузке конфигурационного файла: {e}')
    gs = {}

```
